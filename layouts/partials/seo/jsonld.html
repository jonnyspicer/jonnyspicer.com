{{- /* JSON-LD structured data. Built as dicts and serialised with jsonify so the output is always valid JSON. Every page gets the Person (and the homepage the WebSite); blog posts and essays add a BlogPosting/Article whose author references the Person by @id. */ -}} {{- $personID := printf "%s#person" site.Home.Permalink -}} {{- $sameAs := slice -}} {{- range site.Params.sameAs }}{{ if . }}{{ $sameAs = $sameAs | append . }}{{ end }}{{ end -}} {{- $person := dict "@type" "Person" "@id" $personID "name" site.Params.author "url" site.Home.Permalink "description" site.Params.description -}} {{- with site.Params.personImage }}{{ $person = merge $person (dict "image" (. | absURL)) }}{{ end -}} {{- with $sameAs }}{{ $person = merge $person (dict "sameAs" .) }}{{ end -}} {{- $graph := slice $person -}} {{- if .IsHome -}} {{- $graph = $graph | append (dict "@type" "WebSite" "@id" (printf "%s#website" site.Home.Permalink) "name" site.Title "url" site.Home.Permalink "inLanguage" "en-GB" "author" (dict "@id" $personID) "publisher" (dict "@id" $personID) ) -}} {{- end -}} {{- if and .IsPage (in (slice "blog" "essay") .Section) (ne .Layout "latest") -}} {{- $canonical := partial "seo/canonical.html" . -}} {{- $article := dict "@type" (cond (eq .Section "blog") "BlogPosting" "Article") "@id" (printf "%s#article" $canonical) "headline" .Title "description" (partial "seo/description.html" .) "datePublished" (.Date.Format "2006-01-02T15:04:05Z07:00") "dateModified" (.Lastmod.Format "2006-01-02T15:04:05Z07:00") "author" (dict "@id" $personID) "publisher" (dict "@id" $personID) "url" $canonical "mainEntityOfPage" $canonical "image" ((.Params.image | default site.Params.ogimage) | absURL) "inLanguage" "en-GB" -}} {{- with .WordCount }}{{ $article = merge $article (dict "wordCount" .) }}{{ end -}} {{- $keywords := (.Params.categories | default slice) | append (.Params.tags | default slice) -}} {{- with $keywords }}{{ $article = merge $article (dict "keywords" (delimit . ", ")) }}{{ end -}} {{- $graph = $graph | append $article -}} {{- end -}}
